@Configuration
public class SwaggerConfig {

  @Value("${Swagger.enabled}")
  private boolean enabled; 

  @Bean // 把方法返回的对象注册为Bean
  public class OpenApiConfig {
    return new OpenApi()
                 .info(new Info()
                             .title("系统接口文档")
                             .version("1.0.0")
                             .description("后端 API")
                 );
  }
}

@GetMapping("/{id}")
public Result<UserVO> getById(
  @Parameter(description = "用户ID", example = "1", required = true) // 说明参数 
  @PathVariable Long id) {

}
