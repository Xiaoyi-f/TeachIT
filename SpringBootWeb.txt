将Service及Dao层的实现类, 交给IOC容器管理 
在实现类加上 @Component 注解, 代表把当前类产生的对象交给IOC容器管理 --> 非三层架构层级时候
三层架构层级对应加上 @Controller @Service @Repository 代表有对应的标识的将当前类产生的对象交给IOC容器管理 
参数 value="beanName" 不设置则默认Bean名称为类名将第一个字母改小写后的名称 
注解声明Bean之后不一定会生效, 还需要被组件扫描注解 @ComponentScan 扫描 --> 包含在启动类声明注解 @SpringBootApplication 中, 默认扫描的范围是启动类所在包及其子包

// mybatis-config.xml 
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE configuration
        PUBLIC "-//mybatis.org//DTD Config 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-config.dtd">
<configuration>
    <!-- 1. properties：加载外部properties配置文件（数据库连接信息） -->
    <!-- db.properties 里面写：driver、url、username、password -->
    <properties resource="db.properties"/>

    <!-- 2. settings：全局设置，Mybatis的开关、功能配置 -->
    <settings>
        <!-- 开启驼峰命名自动映射：数据库 create_time → 实体 createTime 【高频】 -->
        <setting name="mapUnderscoreToCamelCase" value="true"/>
        <!-- 日志输出SQL，开发环境打开，看执行的SQL语句 -->
        <setting name="logImpl" value="SLF4J"/>
    </settings>

    <!-- 3. typeAliases：给实体类起别名，简化Mapper里resultType书写 -->
    <typeAliases>
        <!-- 单个起别名 -->
        <typeAlias type="com.learn.pojo.User" alias="User"/>
        <!-- 包扫描：扫描整个包下所有实体，别名默认是类名首字母小写 user -->
        <package name="com.learn.pojo"/>
    </typeAliases>

    <!-- 4. environments：配置数据库环境，可以多套环境（开发/测试/生产） -->
    <!-- default 指定当前使用哪一套环境id -->
    <environments default="dev">
        <environment id="dev">
            <!-- 事务管理器：JDBC事务，交给JDBC管理事务 -->
            <transactionManager type="JDBC"/>
            <!-- dataSource 数据源，连接池 -->
            <dataSource type="POOLED">
                <property name="driver" value="${driver}"/>
                <property name="url" value="${url}"/>
                <property name="username" value="${username}"/>
                <property name="password" value="${password}"/>
            </dataSource>
        </environment>
    </environments>

    <!-- 5. mappers：注册Mapper映射文件！告诉mybatis去哪里找Mapper.xml -->
    <mappers>
        <!-- 方式1：加载单个mapper文件 -->
        <mapper resource="mapper/UserMapper.xml"/>
        <!-- 方式2：包扫描（常用！要求Mapper接口和Mapper.xml同名、同包） -->
        <!-- <package name="com.learn.mapper"/> -->
    </mappers>
</configuration>

// UserMapper.xml 
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
        PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<!-- namespace：绑定Mapper接口的全限定类名！必须和接口全类名一致 -->
<mapper namespace="com.learn.mapper.UserMapper">

    <!-- resultMap：自定义结果映射，数据库字段 -> 实体属性，解决字段名和属性名不一致 -->
    <resultMap id="UserResultMap" type="User">
        <!-- id：主键映射；column数据库列名，property实体属性 -->
        <id column="id" property="id"/>
        <result column="user_name" property="userName"/>
        <result column="price" property="price" jdbcType="DECIMAL"/>
    </resultMap>

    <!-- sql片段：抽取公共SQL，<include refid/>复用 -->
    <sql id="user_base_column">
        id,user_name,price,create_time
    </sql>

    <!-- select 查询标签
         id：对应Mapper接口里面抽象方法名！必须一模一样
         resultType：返回实体类型（简单映射用）
         resultMap：返回自定义resultMap（字段不匹配时用）
         parameterType：传入参数类型（可省略）
    -->
    <select id="selectById" resultType="User">
        SELECT <include refid="user_base_column"/>
        FROM user
        WHERE id = #{id}
    </select>

    <!-- insert 新增 -->
    <insert id="insertUser">
        INSERT INTO user(user_name,price)
        VALUES(#{userName}, #{price})
    </insert>

    <!-- update 修改 -->
    <update id="updateUser">
        UPDATE user
        SET user_name = #{userName}, price = #{price}
        WHERE id = #{id}
    </update>

    <!-- delete 删除 -->
    <delete id="deleteById">
        DELETE FROM user WHERE id = #{id}
    </delete>
</mapper>

/* mybatis-plus */
public interface UserMapper extends BaseMapper<User> {}
// BaseMapper 中 提供好 CRUD 方法 

// 无注解默认情况, 实体类名大驼峰自动转数据库下划线, 名为 id 字段作为主键, 属性名小驼峰自动转数据库下划线
@TableName("table_name")  
@TableId(value="id", type=IdType.AUTO) // 交由数据库自动生成 -ASSIGN_ID-> 雪花算法 
@TableField("plain_field") 
// 注意 MP 反射解析时 会自动去掉字段 is 前缀, 带有 is 前缀的字段 注意使用注解避免出现问题 
@TableField(exist=false) // 不需要作为数据库字段, 不进行识别, 数据库没有对应的字段 

/**
 * 项目配置之 mybatis-plus 配置
 * mybatis-plus:
 *   # 自动扫描对应的包下所有实体类, XML 里面可以省略前缀包名 
 *   type-aliases-package: com.xxx.xxx.xx  # 别名扫描包 --> 给实体类起简短别名  
 *   configuration: 
 *     map-underscore-to-camel-case: true

QueryWrapper<User> qw = new QueryWrapper<>(); 

// Lambda版
LambdaQueryWrapper<User> lqw = new LambdaQueryWrapper<>();

// --> Update 批量/复杂更新
// 普通版 - 字符串硬编码 
UpdateWrapper<User> uw = new UpdateWrapper<>();

LambdaUpdateWrapper luw = new LambdaUpdateWrapper<>();
 
// 自定义SQL eg -> 
List<long> ids = List.of(1L, 2L, 4L);
int amount = 200; 
// 1.构建条件 
LambdaQueryWrapper<User> wrapper = new LambdaQueryWrapper<User>().in(user -> user.getId(), ids);
// 2.自定义SQL方法调用 
userMapper.updateBalanceByIds(wrapper, amount);
// 3.Mapper接口声明 
void updateBalanceByIds(@Param("ew") LambdaQueryWrapper<User> wrapper, @Param("amount") int amount);
// 4.自定义 XML SQL 
<update id="updateBalanceByIds">
  UPDATE tb_user SET balance = balance - #{amount} ${ew.customSqlSegment}
</update>
