import concurrent 
import requests 
import os 

urls = ["https://www.baidu.com", "https://www.bing.com"]

def worker(url):
  response = requests.get(url, timeout=5)
  return response.status_code 

with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor: 
  futures = [executor.submit(worker, url) for url in urls]
  for future in concurrent.futures.as_completed(futures):
    print(future.result())

with concurrent.futures.ProcessPoolExecutor(max_workers=os.cpu_count() or 1) as executor:
  results = executor.map(worker, urls) 
  for result in results: 
    print(result)

import threading 

task = threading.Thread(target=worker, args=(urls[0],), daemon=True)
task.start()
task.join()

lock = threading.Lock()
with lock:
  print("lock using")

import multiprocessing 

process = multiprocessing.Process(target=worker, args=(urls[0],), daemon=True) 
process.start()
process.join() 

lock = multiprocessing.Lock()

  print("lock using")

import asyncio 

async def worker(url):
  response = await asyncio.to_thread(requests.get, url, timeout=5)
  return response.status_code 

async def main():
  tasks = [
    asyncio.create_task(worker("www.baidu.com"))
    asyncio.create_task(worker("www.bing.com"))
  ]

  # task.cancel() --> CancelledError 
  res = await asyncio.gather(*tasks) 
  print(res)

asyncio.run(main())

# oss2 + uuid 
import oss2
import uuid 

class OSSManager:
  def __init__(self):
    self.auth = oss2.Auth(
      "access_key_id",
      "access_key_secret"
    )

    self.bucket = oss2.Bucket(
      self.auth,
      "endpoint",
      "bucket_name"
    )

    # 或自定义域名 
    self.domain = "https://bucket_name.endpoint"

  def upload_file(self, file_content, filename, folder="uploads"):
    extension = filename.rsplit(".", 1)[-1] if "." in filename else ""  
    new_name = f"{uuid.uuid4().hex}.{extension}" if extension else uuid.uuid4().hex 
    oss_key = f"{folder}/{new_name}"

    self.bucket.put_object(oss_key, file_content) 
    return f"{self.domain}/{oss_key}"


  def delete_file(self, url):
    if not url:
      return 

    oss_key = url.replace(f"{self.domain}/", "")
    self.bucket.delete_object(oss_key)
    return True 


  def file_exists(self, oss_key):
    return self.bucket.object_exists(oss_key)

import redis 

# redis原生数据类型: none string list set zset hash stream [原生没有数字和布尔类型, 但是Redis 看到它是整数时, 会用 int 编码来省内存, 布尔类型需要自己约定]
# redis的int是string的底层内部编码, 不为redis的原生数据类型 

REDIS_CONFIG = {
  "host": "127.0.0.1",
  "port": 6379,
  "db": 0, # 0业务 1缓存 2消息队列 3限流 4榜单 5验证码 默认有16个逻辑数据库
  "password": "password",
  "decode_responses": True,  # 底层使用二进制流,开启自动解码
  "socket_timeout": 5,  # 设置超时报错,避免卡死
  "max_connections": 20  # 连接池最大上限 
}

# 可以建立多个cli客户端
pool = redis.ConnectionPool(**REDIS_CONFIG) 
redis_cli = redis.Redis(connection_pool=pool)

# ex过期时间 nx仅当key不存在才设置 
redis_cli.set("key", "value", ex=3600, nx=True) 
redis_cli.mset({"key": "value", "key": "value"})
redis_cli.exists("key")
redis_cli.type("key")
redis_cli.persist("key")  # 移除过期时间,变成永久 
redis_cli.rename("old_key", "new_key")
redis_cli.delete("key")

# 若key不存在则以0处理,最后自动存储
redis_cli.incr("key")
redis_cli.decr("key")

# 禁止执行 redis_cli.keys("*") 会找出所有key 阻塞程序 

# 渐进式遍历查找 user: 开头的key 每次扫描100个 可能返回重复结果

# Redis注意问题: 
# 1.缓存穿透(缓存没有记录,数据库也没有记录,多请求一直请求没有的记录,导致数据库被打爆) --> 缓存空值 / 布隆过滤器 
# 2.缓存击穿(某个热点key突然过期,大量并发请求同时发现缓存没了,全部去查数据库导致数据库压力巨大) --> 互斥锁保护,只允许一个访问去访问数据库,其他访问等待缓存再搭建 
# 3.缓存雪崩(大量key在同一时间过期,数据库等流量瞬间暴增,导致Redis集群宕机,数据库奔溃) --> 打散过期时间 + 高可用 + 限流降级

from matplotlib import colormaps 
import matplotlib.colors as mcolors 

cmap = colormaps["Blues"]

def color_func_with_highlight(word, font_size, position, orientation, random_state, **kwargs):
  words = kwargs.get("words", {})
  if not words:
    return mcolors.to_hex(cmap(random_state.rand()))

  max_word = max(words, key=words.get) 

  if word == max_word:
    return "red" 

  max_freq = words[max_word]
  min_freq = min(words.values())
  if max_freq > min_freq:
    norm_freq = (words[word] - min_freq) / (max_freq - min_freq)
  else:
    norm_freq = 0.5 

  rgba = cmap(norm_freq) 
  return mcolors.to_hex(rgba)

w.recolor(color_func=color_func_with_highlight)

file = open(r"file.txt", encoding="utf-8") 
string = file.read() 
file.close() 

w.generate(string)
w.to_file("file.jpg")
