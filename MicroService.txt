public class DiscoveryTest {

  @Autowired 
  DiscoveryClient discoveryClient; 

  @Test 
  void discoveryClient() {
    for (String service : discoveryClient.getServices()) {
      System.out.println("service =" + service);
      List<ServiceInstance> instances = discoveryClient.getInstances(service); 
      for (ServiceInstance instance : instances) {
        System.out.println("ip:" + instance.getHost() + ";" + "port = " + instance.getPort());
      }

      // 负载均衡 挑选实例 -- 简单演示 
      ServiceInstance instance = instances.get(RandomUtil.randomInt(instances.size()));

      // 请求 -- 远程调用 
      ResponseEntity<List<ItemDTO>> response = restTemplate.exchange(
        instance.getUri() + "/items?ids={ids}",
        HttpMethod.GET,
        null, // 请求实体 -- 请求头和请求体 
        new ParameterizedTypeReference<List<ItemDTO>>() {}, // 返回值类型
        CollUtils.join(itemIds, ",") // Hutool 工具 将集合拼接为以对应字符(串)为分隔符的字符串 
        );

      // 处理结果 
      List<ItemDTO> items = null;
      if (response.getStatusCode().is2xxSuccessful()) {
        items = response.getBody();
      }
    }
  }
}

@FeignClient("item-service", url = "http://第三方服务地址域名") // 请求第三方服务地址传递的名字参数可以随意填写  
public interface ItemClient {

  @GetMapping("/items") // OpenFeign 利用动态代理帮我们实现这个方法 自动向 http://item-service/items发送GET请求, 携带ids为请求参数, 自动将返回值处理为 List<ItemDTO> 
  List<ItemDTO> queryItemByIds(@RequestParam("ids") Collection<Long> ids); // 请求头/请求参数等同理可用 
}

public class CartServiceImpl extends ServiceImpl<CartMapper, Cart> implements ICartService {

  private ItemClient itemClient; 

  @Override 
  public void addItem2Cart(CartFormDTO cartFormDTO) { ... }

  @Override 
  public List<CartDTO> queryMyCarts() { ... }

  private void handleCartItems(List<CartVO> vos) {
    Set<Long> itemIds = vos.stream().map(CartVO::getItemId).collect(Collectors.toSet()); 
    List<ItemDTO> items = itemClient.queryItemByIds(itemIds);
    if (CollUtils.isEmpty(items)) {
      throw new BadRequestException("购物车中商品不存在!");
    }
  }
}
